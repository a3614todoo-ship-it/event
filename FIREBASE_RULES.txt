rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // 1. 活動資料
    match /events/{eventId} {
      allow read: if true;
      allow write: if true;
    }
    
    // 2. 報名資料
    match /event_registrations/{regId} {
      allow create: if 
        request.resource.data.eventId is string &&
        request.resource.data.userEmail.size() > 5 &&
        request.resource.data.userName.size() > 0 &&
        (
          request.resource.data.status == 'registered' ||
          request.resource.data.status == 'waiting' ||
          request.resource.data.status == 'pending_payment'
        );
      
      allow read: if true;
      allow update, delete: if true;
    }

    // 3. 問卷回饋資料
    match /event_surveys/{surveyId} {
      allow create: if true;
      allow read: if true;
      allow update, delete: if true;
    }
     
    // 4. 管理員登入資料
    match /admin_access/{docId} {
      allow read, write: if true;
    }
  }
}
